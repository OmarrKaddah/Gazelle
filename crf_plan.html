def token_features(tokens, i, gazetteer): tok = tokens[i] feats = { 'pos': tok['pos'], 'prefix2': tok['word'][:2], 'suffix2': tok['word'][-2:], 'char_len': len(tok['word']), 'has_definite_art': tok['word'].startswith('ال'), 'has_prep_clitic': tok['word'][0] in 'بلكف', 'is_arabic_num': bool(re.match(r'^[\u0660-\u0669]+', tok['word'])), 'in_gazetteer': tok['word'] in gazetteer, 'root': tok.get('root', ''), # from Farasa } # window features if i > 0: feats['-1:pos'] = tokens[i-1]['pos'] feats['-1:word'] = tokens[i-1]['word'] if i < len(tokens)-1: feats['+1:pos'] = tokens[i+1]['pos'] feats['+1:word'] = tokens[i+1]['word'] return feats
