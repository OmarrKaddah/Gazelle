the CRF training table — one row = one token

Key concept: GLiNER gives you spans (whole phrases). The CRF works on tokens (individual words). The bridge is BIO alignment: you map each span back onto the tokenised text and assign B- (beginning), I- (inside), or O (outside) to each token. Each row in the training table = one token with its features and its BIO label as the target.

sentence: كتاب نائب محافظ البنك المركزي المصري بتاريخ ١٢ يوليو ٢٠٢٣
what the target column (BIO label) looks like
BIO scheme
# B = Beginning of entity span B-RegulatoryBody → first token of البنك المركزي # I = Inside entity span (continuation) I-RegulatoryBody → المركزي, المصري # O = Outside (not an entity) O → كتاب, نائب, بتاريخ
BIOES scheme (recommended)
# B = Beginning, I = Inside # E = End, S = Single token entity S-Date → ٢٠٢٣ (single-token date) B-REG → البنك I-REG → المركزي E-REG → المصري # E-tag = explicit end → CRF learns # span boundaries more precisely

step 2 — clean spans and apply correction rules
CLAUSE_REF = re.compile(r'^[\u0660-\u0669\d]+-[\u0660-\u0669\d]+-[\u0660-\u0669\d]+$') MONTH_AR = {'يناير','فبراير','مارس','أبريل','مايو','يونيو', 'يوليو','أغسطس','سبتمبر','أكتوبر','نوفمبر','ديسمبر'} TYPE_MAP = { 'BankingInstitution': 'BankingInstitution', 'RegulatoryBody': 'RegulatoryBody', 'FinancialInstrument':'BankingInstitution', # review each 'MonetaryAmount': 'MonetaryAmount', 'Date': 'Date', 'Law': 'LegalReference', 'License': 'LegalReference', 'Person': 'PersonRole', 'Document': None, # drop 'RegulatoryRequirement': None, # drop } def clean_entity(ent): text = ent['text'].replace('\n', ' ').strip() etype = TYPE_MAP.get(ent['type']) if etype is None: return None # drop if CLAUSE_REF.match(text): return None # not a date if text.startswith('وزارة'): etype = 'GovernmentBody' if 'البنك المركزي' in text: etype = 'RegulatoryBody' return {**ent, 'text': text, 'type': etype}
step 3 — tokenise and BIO-align
from farasa.segmenter import FarasaSegmenter segmenter = FarasaSegmenter(interactive=True) def bio_align(chunk_text, entities): """ Returns list of (token, char_start, char_end, bio_label) """ tokens = [] pos = 0 for seg in segmenter.segment(chunk_text).split(): start = chunk_text.find(seg, pos) end = start + len(seg) tokens.append({'w': seg, 's': start, 'e': end, 'label': 'O'}) pos = end for ent in sorted(entities, key=lambda x: x['start']): first = True for tok in tokens: if ent['start'] <= tok['s'] < ent['end']: tok['label'] = ('B-' if first else 'I-') + ent['type'] first = False return tokens
step 4 — extract features for each token
ORG_TRIGGERS = {'بنك','شركة','هيئة','مؤسسة','وزارة','لجنة','سلطة','مجلس'} CURRENCY = {'جنيه','دولار','يورو','مليار','مليون','ألف'} MONTHS = {'يناير','فبراير','مارس','أبريل','مايو','يونيو', 'يوليو','أغسطس','سبتمبر','أكتوبر','نوفمبر','ديسمبر'} ORG_GAZ = {'البنك المركزي المصري','البنك المركزي','الجهاز المصرفي'} LOC_GAZ = {'جمهورية مصر العربية','مصر','القاهرة'} def extract_features(tokens, i): w = tokens[i]['w'] pre = tokens[i-1]['w'] if i > 0 else '__BOS__' nxt = tokens[i+1]['w'] if i < len(tokens)-1 else '__EOS__' return { # --- surface --- 'token': w, 'prefix_2': w[:2], 'suffix_2': w[-2:], 'suffix_3': w[-3:], 'token_len': len(w), # --- Arabic morphology --- 'has_def_art': w.startswith('ال'), 'has_prep_clitic': w[0] in 'بلكفو' and len(w) > 2, 'is_arabic_num': bool(re.match(r'^[\u0660-\u0669]+', w)), 'is_western_num': w.isdigit(), 'contains_latin': bool(re.search(r'[a-zA-Z]', w)), 'is_clause_ref': bool(CLAUSE_REF.match(w)), # --- gazetteers --- 'in_org_gaz': w in ORG_GAZ, 'in_loc_gaz': w in LOC_GAZ, 'in_currency': w in CURRENCY, 'in_month': w in MONTHS, # --- context window --- 'prev_word': pre, 'next_word': nxt, 'prev_is_trigger': pre in ORG_TRIGGERS, 'next_is_currency': nxt in CURRENCY, 'is_sent_start': i == 0, # char n-grams **{f'c2_{ng}': True for ng in [w[j:j+2] for j in range(len(w)-1)]}, **{f'c3_{ng}': True for ng in [w[j:j+3] for j in range(len(w)-2)]}, }
step 5 — assemble training data and train CRF
import sklearn_crfsuite from sklearn_crfsuite import metrics X_train, y_train = [], [] for chunk in all_chunks: # your 200 annotated chunks entities = [clean_entity(e) for e in by_chunk[chunk['chunkId']]] entities = [e for e in entities if e] # drop None tokens = bio_align(chunk['text'], entities) feats = [extract_features(tokens, i) for i in range(len(tokens))] labels= [t['label'] for t in tokens] X_train.append(feats) y_train.append(labels) crf = sklearn_crfsuite.CRF( algorithm='lbfgs', c1=0.05, c2=0.1, max_iterations=150, all_possible_transitions=True ) crf.fit(X_train, y_train) # evaluate on held-out 20 chunks y_pred = crf.predict(X_test) print(metrics.flat_classification_report(y_test, y_pred, digits=3))
